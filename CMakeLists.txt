cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VITASDK})
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain")
 endif()
 project(vita_hw_audio_decoder VERSION 1.0.0 LANGUAGES C)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET VitaAudioDecoder::Interface)
 add_subdirectory(interface)
endif()
add_library(vita_hw_audio_decoder STATIC src/vita_hw_audio_decoder.c src/module_log.c)
add_library(VitaHwAudioDecoder::VitaHwAudioDecoder ALIAS vita_hw_audio_decoder)
set_target_properties(vita_hw_audio_decoder PROPERTIES EXPORT_NAME VitaHwAudioDecoder VERSION 1.0.0)
target_include_directories(vita_hw_audio_decoder PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(vita_hw_audio_decoder PUBLIC c_std_11)
target_compile_options(vita_hw_audio_decoder PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -O3 -ffunction-sections -fdata-sections)
target_link_libraries(vita_hw_audio_decoder PUBLIC VitaAudioDecoder::Interface SceAudiodec_stub SceSysmodule_stub SceSysmem_stub)
install(TARGETS vita_hw_audio_decoder EXPORT VitaHwAudioDecoderTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/vita_hw_audio_decoder.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT VitaHwAudioDecoderTargets NAMESPACE VitaHwAudioDecoder:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaHwAudioDecoder)
configure_package_config_file(cmake/VitaHwAudioDecoderConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/VitaHwAudioDecoderConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaHwAudioDecoder)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/VitaHwAudioDecoderConfigVersion.cmake VERSION 1.0.0 COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/VitaHwAudioDecoderConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/VitaHwAudioDecoderConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaHwAudioDecoder)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DEPENDENCIES.lock DESTINATION share/licenses/vita-hw-audio-decoder)
